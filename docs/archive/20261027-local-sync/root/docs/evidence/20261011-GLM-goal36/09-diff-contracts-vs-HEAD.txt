+def plan_deployment_outcome_error(
+    deliverable: str, steps: list[str], acceptance_checks: list[str]
+) -> str | None:
+    """Reject explicit "source presence proves deployment/runtime" conclusions.
+
+    This is a narrow causal/outcome guard, not a judgment on whether the
+    deployment happened. A match needs an explicit causal connector AND a
+    deployment/runtime conclusion, so "source presence means the file was
+    uploaded / material is locatable" stays allowed. Negative statements,
+    conditional wording and plans that actually produce a runtime record
+    stay allowed; a negation only exempts inside its own clause, so an
+    independent causal assertion after a comma is still rejected. Inserting
+    a bare "运行" into a step cannot rescue a causal acceptance conclusion.
+    """
+    for text in (deliverable, *steps, *acceptance_checks):
+        for sentence in re.split(r"[。！？；;\n]", text):
+            for match in _SOURCE_RUNTIME_CAUSAL.finditer(sentence):
+                prefix = _causal_negation_prefix(sentence, match.start())
+                if _CAUSAL_EXEMPTION.search(prefix) or _CAUSAL_EXEMPTION.search(match.group()):
+                    continue
+                return "AI task plan claims source presence proves a deployment or runtime outcome"
+    for text in (deliverable, *acceptance_checks):
+        for sentence in re.split(r"[。！？；;\n]", text):
+            if _OUTCOME_NEGATION.search(sentence):
+                continue
+            if _DEPLOYMENT_OUTCOME.search(sentence) and not any(
+                _RUNTIME_RECORD_STEP.search(step) for step in steps
+            ):
+                return (
+                    "AI task plan asserts a deployment or runtime outcome "
+                    "without a runtime-record step"
+                )
+    return None
