import collections, hashlib, json, pathlib, re, shutil, subprocess
ROOT=pathlib.Path(r'D:\jbcode\web ai\jianli');OUT=ROOT/'backend/.tmp/github-sync-20261027';WT=ROOT/'backend/.tmp/github-evidence-20261027'
d=json.loads((OUT/'plan.json').read_text(encoding='utf8'));paths=[]
for x in d['files']:
    if x['category'] not in ('evidence','source-archive','delivery'):continue
    src=pathlib.Path(x['worktree'])/x['path'];data=src.read_bytes()
    assert hashlib.sha256(data).hexdigest()==x['sha256'],f'concurrent change: {x["path"]}'
    dest=WT/x['remote_path'];dest.parent.mkdir(parents=True,exist_ok=True)
    if dest.exists():assert dest.read_bytes()==data,str(dest)
    dest.write_bytes(data);paths.append(x['remote_path'])
    # All new historical text payloads were scanned; raw credential logs/reviews excluded.
    if b'\0' not in data[:8192] and len(data)<20_000_000:
        s=data.decode('utf8',errors='replace')
        assert not re.search(r'\bsk-[A-Za-z0-9_-]{20,}|-----BEGIN (?:RSA |OPENSSH |EC )?PRIVATE KEY-----|\beyJ[A-Za-z0-9_-]{15,}\.[A-Za-z0-9_-]{15,}\.',s),x['path']
base=WT/'docs/archive/20261027-local-sync'
base.mkdir(parents=True,exist_ok=True)
(base/'.gitattributes').write_text('* -text\n**/* -text\n',encoding='utf8');paths.append('docs/archive/20261027-local-sync/.gitattributes')

完整分类、排除清单、提交位置、安全边界和本轮日志见 [同步审查入口](../../evidence/20261027-github-sync/README.md)。原有失败、报告与人工原件未被改写。

(base/'README.md').write_text(readme,encoding='utf8');paths.append('docs/archive/20261027-local-sync/README.md')
logdir=WT/'docs/evidence/20261027-github-sync/logs';logdir.mkdir(parents=True,exist_ok=True)
for p in OUT.iterdir():
    if p.name.endswith(('.log','.xml')) or p.name in [n+'.json' for n in ['backend-target','ruff','frontend-tests','frontend-lint','frontend-typecheck','frontend-build','runner-test']]:
        dest=logdir/p.name;shutil.copyfile(p,dest);paths.append(dest.relative_to(WT).as_posix())

subprocess.run(['git','-c','core.longpaths=true','add','-f','--pathspec-from-file='+str(f),'--pathspec-file-nul'],cwd=WT,check=True,capture_output=True)
staged=set(filter(None,subprocess.check_output(['git','diff','--cached','--name-only','-z'],cwd=WT).decode('utf8').split('\0')))
assert staged==set(paths)
# Check the actual staged bytes, not only the source snapshot.
entries={}
for item in subprocess.check_output(['git','ls-files','--stage','-z'],cwd=WT).decode('utf8').split('\0'):
    if not item:continue
    meta,path=item.split('\t',1);entries[path]=meta.split()[1]
assets=[x for x in d['files'] if x['category'] in ('evidence','source-archive','delivery')]
ids=[entries[x['remote_path']] for x in assets]

pos=0
for x in assets:

    blob=raw[pos:pos+size];pos+=size+1
    assert hashlib.sha256(blob).hexdigest()==x['sha256'],x['path']
